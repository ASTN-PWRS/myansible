CREATE TABLE pdf_pages (

3. pdf_chunks（意味チャンク＋bbox画像URL）
Gemini Thinking が返す意味チャンクを保存し、

CREATE TABLE pdf_chunks (
  id              bigserial PRIMARY KEY,
  doc_id          uuid NOT NULL,
  page_number     int NOT NULL,
  chunk_index     int NOT NULL,
  text            text,
  text_tsv        tsvector,
  chunk_type      text,
  bbox            jsonb,
  bbox_image_url  text,
  embedding       vector(768),
  meta            jsonb,
  created_at      timestamptz DEFAULT now()
);

-- chunk の一意性保証
CREATE UNIQUE INDEX idx_chunks_unique
  ON pdf_chunks (doc_id, page_number, chunk_index);

-- 検索用
CREATE INDEX idx_chunks_tsv ON pdf_chunks USING GIN (text_tsv);
CREATE INDEX idx_chunks_embedding_hnsw ON pdf_chunks USING hnsw (embedding);

Gemini Vision メタプロンプト（完全版）

あなたは高精度なドキュメント解析エンジンです。
入力される画像は PDF のページ画像です。

必ず以下の処理を行い、指定の JSON 形式で返してください。

【必須タスク】
1. OCR（日本語＋英語）
   - テキストを正確に抽出する
   - 誤字補正は行わない（原文忠実）

2. bbox 抽出
   - テキストブロックごとに bbox を返す
   - 図・表・写真などの非テキスト領域も bbox を返す
   - 座標は画像のピクセル基準で {x, y, w, h}

3. 図表分類
   - 各領域を以下のいずれかに分類する
     "text", "figure", "table", "photo", "caption", "other"

4. キャプション抽出
   - 図や表に付随するキャプションを抽出し、対応する領域と紐付ける

5. ページ構造化
   - ページ内の領域を上から下へ並べる
   - セクションタイトルがあれば抽出する

【出力形式（厳密）】
{
  "page_number": <int>,
  "width": <int>,
  "height": <int>,
  "blocks": [
    {
      "block_index": <int>,
      "type": "text" | "figure" | "table" | "photo" | "caption" | "other",
      "bbox": { "x": <int>, "y": <int>, "w": <int>, "h": <int> },
      "text": "<string or empty>",
      "caption_for": <block_index or null>
    }
  ]
}

注意：

必ず JSON のみを返す

余計な説明文は返さない

bbox は整数

テキストは改行を保持

import psycopg2

import requests

# =========================
# 設定

VISION_API_URL = "https://internal.api.local/vision/analyze"
THINKING_API_URL = "https://internal.api.local/thinking/chunk"
TOKEN_URL = "https://internal.api.local/auth/token"

PDF_DIR = Path("./pdf-originals")
PAGE_IMG_DIR = Path("./data/pdf-pages")
BBOX_IMG_DIR = Path("./data/pdf-bbox")

PAGE_IMG_DIR.mkdir(parents=True, exist_ok=True)
BBOX_IMG_DIR.mkdir(parents=True, exist_ok=True)

PG_CONN_INFO = "dbname=yourdb user=youruser password=yourpass host=localhost port=5432"


# =========================

def get_token():
    """
    社内APIのtokenを取得するダミー実装
    """
    resp = requests.post(TOKEN_URL, json={"client_id": "xxx", "client_secret": "yyy"})
    return resp.json()["access_token"]

# DB ヘルパ
# =========================

def get_pg_conn():
    return psycopg2.connect(PG_CONN_INFO)


def init_tables():
    sqls = [
        """
        CREATE TABLE IF NOT EXISTS pdf_documents (
          doc_id      text PRIMARY KEY,
          file_name   text,
          file_url    text,
          title       text,
          page_count  int,
          created_at  timestamptz DEFAULT now()
        );
        """,
        """
        CREATE TABLE IF NOT EXISTS pdf_pages (
          doc_id          text,

          bbox_image_url  text,
          embedding       vector(768),

    ]
    conn = get_pg_conn()
    cur = conn.cursor()
    for s in sqls:
        cur.execute(s)
    conn.commit()
    cur.close()
    conn.close()


# =========================
# Vision API 呼び出し
# =========================

def call_vision_api(page_image_path: Path, page_number: int, token: str) -> dict:
    with open(page_image_path, "rb") as f:
        img_b64 = base64.b64encode(f.read()).decode()

    payload = {
        "page_number": page_number,
        "image": img_b64
    }

    resp = requests.post(
        VISION_API_URL,
        headers={"Authorization": f"Bearer {token}"},
        json=payload
    )
    return resp.json()


# =========================
# Thinking API 呼び出し
# =========================

def call_thinking_api(doc_id: str, vision_json: dict, token: str) -> dict:
    payload = {
        "doc_id": doc_id,
        "vision": vision_json
    }

    resp = requests.post(
        THINKING_API_URL,
        headers={"Authorization": f"Bearer {token}"},
        json=payload
    )
    return resp.json()


# =========================
# 画像保存 & bbox切り出し
# =========================

def save_page_image(doc_id: str, page_number: int, pil_image: Image.Image) -> Path:
    out_dir = PAGE_IMG_DIR / doc_id
    out_dir.mkdir(parents=True, exist_ok=True)
    out_path = out_dir / f"page_{page_number}.png"
    pil_image.save(out_path)
    return out_path


def crop_bbox_image(doc_id: str, page_number: int, chunk_index: int,
                    page_image_path: Path, bbox: dict) -> Path:
    img = Image.open(page_image_path)
    x, y, w, h = bbox["x"], bbox["y"], bbox["w"], bbox["h"]
    crop = img.crop((x, y, x + w, y + h))

    out_dir = BBOX_IMG_DIR / doc_id
    out_dir.mkdir(parents=True, exist_ok=True)
    out_path = out_dir / f"page_{page_number}_chunk_{chunk_index}.png"
    crop.save(out_path)
    return out_path


def local_url_for(path: Path) -> str:
    base = "http://localhost:8000"
    rel = path.relative_to(Path("./data"))
    return f"{base}/{rel.as_posix()}"


# =========================
# embedding（ダミー）
# =========================

def embed_text(text: str):
    return [0.0] * 768


# =========================
# PDF → Vision → Thinking → PostgreSQL
# =========================

def process_pdf(pdf_path: Path):
    token = get_token()

    doc_id = pdf_path.stem
    file_name = pdf_path.name
    file_url = f"http://localhost:8000/pdf-originals/{file_name}"

    pages = convert_from_path(str(pdf_path))
    page_count = len(pages)

    conn = get_pg_conn()
    cur = conn.cursor()

    # pdf_documents 登録
    cur.execute(
        """
        INSERT INTO pdf_documents (doc_id, file_name, file_url, title, page_count)
        VALUES (%s, %s, %s, %s, %s)
        ON CONFLICT (doc_id) DO UPDATE
        SET file_name = EXCLUDED.file_name,
            file_url  = EXCLUDED.file_url,
            page_count = EXCLUDED.page_count;
        """,
        (doc_id, file_name, file_url, doc_id, page_count)
    )

    # 各ページ処理
    for i, pil_page in enumerate(pages, start=1):
        page_img_path = save_page_image(doc_id, i, pil_page)
        page_image_url = local_url_for(page_img_path)

        # Vision API
        vision_json = call_vision_api(page_img_path, i, token)

        # Thinking API
        thinking_json = call_thinking_api(doc_id, vision_json, token)

        # pdf_pages 登録
        cur.execute(
            """
            INSERT INTO pdf_pages (doc_id, page_number, page_image_url,
                                   summary, has_figure, has_table, has_photo, sections)
            VALUES (%s, %s, %s, %s, %s, %s, %s, %s)
            ON CONFLICT (doc_id, page_number) DO UPDATE
            SET page_image_url = EXCLUDED.page_image_url;
            """,
            (
                doc_id,
                i,
                page_image_url,
                None,
                None,
                None,
                None,
                json.dumps({})
            )
        )

        # pdf_chunks 登録
        chunks = thinking_json["chunks"]
        rows = []
        for chunk in chunks:
            chunk_index = chunk["chunk_index"]
            chunk_type = chunk["chunk_type"]
            text = chunk.get("text") or ""
            bbox = chunk.get("bbox") or {"x": 0, "y": 0, "w": 0, "h": 0}
            meta = chunk.get("meta") or {}

            bbox_img_path = crop_bbox_image(
                doc_id, i, chunk_index, page_img_path, bbox
            )
            bbox_image_url = local_url_for(bbox_img_path)

            emb = embed_text(text)

            rows.append((
                doc_id,
                i,
                chunk_index,
                text,
                chunk_type,
                json.dumps(bbox),
                bbox_image_url,
                emb,
                json.dumps(meta),
            ))

        if rows:
            execute_values(
                cur,
                """
                INSERT INTO pdf_chunks
                  (doc_id, page_number, chunk_index,
                   text, chunk_type, bbox, bbox_image_url,
                   embedding, meta)
                VALUES %s
                """,
                rows
            )

    conn.commit()
    cur.close()
    conn.close()


# =========================
# メイン
# =========================

if __name__ == "__main__":
    init_tables()

    for pdf_file in PDF_DIR.glob("*.pdf"):
        print(f"Processing {pdf_file.name}...")
        process_pdf(pdf_file)
        print(f"Done: {pdf_file.name}")

① Gemma 4 用：検索意図解析プロンプト（検索条件生成）

あなたは高精度な検索条件生成エンジンです。
ユーザーの自然言語による検索要求を受け取り、

【あなたが行う処理】

2. 全文検索用のクエリ（tsvector用）を生成する
3. ベクトル検索用のクエリ（embedding用）を生成する
4. 図表フィルタを推定する

   - 指定がなければ bbox_area_min=0, bbox_area_max=10000000
6. ページ範囲を推定する
   - 指定がなければ [1, 10000]
7. top_k を推定する（指定がなければ 10）

【出力形式（厳密）】
必ず以下の JSON のみを返すこと：

{
  "query": "<全文検索用クエリ>",
  "vector_query": "<ベクトル検索用クエリ>",
  "top_k": <int>,
  "filters": {
    "chunk_type_in": ["text" | "figure" | "table" | "photo" | "caption"],
    "bbox_area_min": <int>,
    "bbox_area_max": <int>,
    "page_range": [<int>, <int>]
  }
}

余計な説明文は絶対に返さない。

🟦 Gemini Vision + Gemini Thinking 統合メタプロンプト

あなたは高精度なドキュメント解析エンジンです。

このプロンプトは 2 種類の処理モードを持ちます。
ユーザーは毎回以下のいずれかを指定します。

- mode: "vision"
- mode: "thinking"

入力は以下の形式です：

{
  "mode": "<vision | thinking>",
  "page_number": <int>,
  "doc_id": "<string>",
  "image": "<base64 or URL or null>",
  "vision_json": "<JSON or null>"
}

------------------------------------------------------------
# 🟦 mode = "vision" の場合（画像 → 構造化 JSON）
------------------------------------------------------------

あなたは Gemini Vision として動作します。
入力される画像は PDF のページ画像です。

必ず以下の処理を行い、指定の JSON 形式で返してください。

【必須タスク】
1. OCR（日本語＋英語）
   - 原文忠実、誤字補正なし

2. bbox 抽出
   - テキストブロックごとに bbox
   - 図・表・写真など非テキスト領域も bbox
   - {x, y, w, h} は整数

3. 図表分類
   - "text", "figure", "table", "photo", "caption", "other"

4. キャプション抽出
   - 図表に紐付ける

5. ページ構造化
   - 上から下へ並べる
   - セクションタイトルがあれば抽出

【出力形式（厳密）】
{
  "page_number": <int>,
  "width": <int>,
  "height": <int>,
  "blocks": [
    {
      "block_index": <int>,
      "type": "text" | "figure" | "table" | "photo" | "caption" | "other",
      "bbox": { "x": <int>, "y": <int>, "w": <int>, "h": <int> },
      "text": "<string or empty>",
      "caption_for": <block_index or null>
    }
  ]
}

必ず JSON のみを返す。

------------------------------------------------------------
# 🟧 mode = "thinking" の場合（構造化 JSON → 意味チャンク化）
------------------------------------------------------------

あなたは Gemini Thinking として動作します。
入力は Gemini Vision が返した JSON です。

以下の処理を行い、指定の JSON 形式で返してください。

【必須タスク】
- 意味チャンク化（semantic chunking）
- 図・表・写真は単独チャンク
- キャプションは対応する図表チャンクに紐付ける

- チャンク分類
  "text", "figure", "table", "photo", "caption"

- メタデータ生成
  - セクションタイトル
  - 図表番号（推定）
  - キャプション関連付け
  - ページ内順序

- DB登録用データ生成

【出力形式（厳密）】
{
  "doc_id": "<string>",
  "page_number": <int>,
  "chunks": [
    {
      "chunk_index": <int>,
      "chunk_type": "text" | "figure" | "table" | "photo" | "caption",
      "text": "<string or empty>",
      "bbox": { "x": <int>, "y": <int>, "w": <int>, "h": <int> },
      "meta": {
        "section": "<string or null>",
        "caption_for": <chunk_index or null>,
        "order": <int>
      }
    }
  ]
}

必ず JSON のみを返す。
